---Question 2.4: . Store Efficiency (Revenue per Transaction per Store) | This measures how much revenue each store generates per transaction.

SELECT 
    store_id,
    store_location,
   round(sum (transaction_qty * CAST(REPLACE(Unit_price,',','.') AS double)) / COUNT(DISTINCT transaction_id),1) AS revenue_per_transaction
FROM bright_coffee.sales.coffee_dataset
GROUP BY store_id, store_location
ORDER BY revenue_per_transaction DESC;

---- Question 8: Calculating the highest revenue product category per time bucket (Morning, Afternoon, Evening, Night):


WITH bucketed_revenue AS (
    SELECT 
        product_category,
        CASE    
            WHEN date_format(transaction_time, 'HH:mm:ss') BETWEEN '06:00:00' AND '11:59:59' THEN 'Morning'
            WHEN date_format(transaction_time, 'HH:mm:ss') BETWEEN '12:00:00' AND '16:59:59' THEN 'Afternoon'
            WHEN date_format(transaction_time, 'HH:mm:ss') BETWEEN '17:00:00' AND '19:59:59' THEN 'Evening'
            ELSE 'Night'
        END AS time_bucket,
        SUM(transaction_qty * CAST(REPLACE(unit_price, ',', '.') AS DECIMAL(10,2))) AS total_revenue
    FROM bright_coffee.sales.coffee_dataset
    GROUP BY product_category,
             CASE    
                 WHEN date_format(transaction_time, 'HH:mm:ss') BETWEEN '06:00:00' AND '11:59:59' THEN 'Morning'
                 WHEN date_format(transaction_time, 'HH:mm:ss') BETWEEN '12:00:00' AND '16:59:59' THEN 'Afternoon'
                 WHEN date_format(transaction_time, 'HH:mm:ss') BETWEEN '17:00:00' AND '19:59:59' THEN 'Evening'
                 ELSE 'Night'
             END
)
SELECT time_bucket, product_category, total_revenue
FROM (
    SELECT time_bucket, product_category, total_revenue,
           ROW_NUMBER() OVER (PARTITION BY time_bucket ORDER BY total_revenue DESC) AS rank
    FROM bucketed_revenue
) ranked
WHERE rank = 1
ORDER BY time_bucket;
